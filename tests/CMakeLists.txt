# Every CXPM/Testing/*Test.hpp header self-registers a TestGroup (see
# libraries/cxpm-interface/src/CXPM/Testing/TestGroup.hpp). This aggregator glob-includes
# all of them into one generated translation unit, exactly as documented in
# docs/testing.md, so accidental helper/name collisions and ODR problems surface at
# compile time instead of hiding behind separate test binaries.
set(CXPM_TEST_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../libraries/cxpm-interface/src")

file(GLOB_RECURSE CXPM_TEST_HEADERS CONFIGURE_DEPENDS "${CXPM_TEST_INCLUDE_ROOT}/CXPM/Testing/*Test.hpp")

set(GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
file(WRITE "${GENERATED_TESTS}" "// Generated by CMake. Every CXPM/Testing/*Test.hpp header self-registers a TestGroup.\n")
foreach(TEST_HEADER IN LISTS CXPM_TEST_HEADERS)
    file(RELATIVE_PATH RELATIVE_HEADER "${CXPM_TEST_INCLUDE_ROOT}" "${TEST_HEADER}")
    file(APPEND "${GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
endforeach()

add_executable(cxpm_tests main.cpp "${GENERATED_TESTS}")
target_link_libraries(cxpm_tests PRIVATE cxpm-interface)
target_compile_features(cxpm_tests PRIVATE cxx_std_23)
if(MSVC)
    target_compile_options(cxpm_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(cxpm_tests PRIVATE -Wall -Wextra)
endif()

if(CXPM_ENABLE_SANITIZERS AND NOT MSVC)
    target_compile_options(cxpm_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cxpm_tests PRIVATE -fsanitize=address,undefined)
endif()

if(CXPM_ENABLE_COVERAGE)
    if(MSVC)
        message(FATAL_ERROR "CXPM_ENABLE_COVERAGE requires gcc or clang")
    endif()
    target_compile_options(cxpm_tests PRIVATE --coverage -O0 -g)
    target_link_options(cxpm_tests PRIVATE --coverage)

    find_program(CXPM_GCOVR_EXECUTABLE gcovr)
    if(CXPM_GCOVR_EXECUTABLE)
        add_custom_target(coverage
            COMMAND ${CMAKE_CTEST_COMMAND} --test-dir "${CMAKE_BINARY_DIR}" -L unit --output-on-failure
            COMMAND ${CMAKE_COMMAND} -E rm -rf "${CMAKE_SOURCE_DIR}/docs/coverage"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_SOURCE_DIR}/docs/coverage"
            COMMAND ${CXPM_GCOVR_EXECUTABLE}
                --root "${CMAKE_SOURCE_DIR}"
                --filter "libraries/cxpm-interface/src/.*"
                --exclude ".*/Testing/.*"
                --exclude ".*/tests/.*"
                --html-details "${CMAKE_SOURCE_DIR}/docs/coverage/index.html"
                --html-title "cxpm coverage"
                --print-summary
                "${CMAKE_BINARY_DIR}"
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
            DEPENDS cxpm_tests
            COMMENT "Running cxpm_tests and generating HTML coverage under docs/coverage"
        )
    else()
        message(WARNING "gcovr not found: `coverage` target will not be available")
    endif()
endif()

add_test(NAME cxpm_tests COMMAND cxpm_tests)
set_tests_properties(cxpm_tests PROPERTIES LABELS "unit")
